## fork
자료는 C언어면서 과제는 왜 파이썬임?
- `os.getpid`
- `os.fork`
```py
os.getpid
pid=os.fork()

```

`pid=fork()`의 결과가 0이면 자식, 그 외는 부모 프로세스


```c
#include <unistd.h>
#include <sys/types.h>
int main() {
	getpid();
	int pid=fork();

	#if pid == 0 -> child process
	#else -> parent process
	if(pid==0) printf("Child pid=%d", getpid());
	else printf("Parent pid=%d", getpid());

	return 0;
}
```

### fork example
```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int main() 
{
  int pid1, pid2;
  pid1 = fork();
  if (pid1 == 0) {
    printf("[Child 1] : Hello, world ! pid=%d\n",getpid());
    exit(0);
  }
  pid2 = fork();
  if (pid2 == 0) {
    printf("[Child 2] : Hello, world ! pid=%d\n",getpid());
    exit(0);
  }
  printf("[PARENT] : Hello, world ! pid=%d\n",getpid());
}

```
### result
```
$ fork3
[Parent] Hello, world! pid=15740
[Child 1] Hello, world! pid=15741
[Child 2] Hello, world! pid=15742
```


## waitpid
```c
int status, pid, child;
child=waitpid(pid,&status,0);

```
```py
pid=os.fork()
child,status=os.waitpid(pid)

```


```
#include <sys/types.h>
#include <sys/wait.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int main() 
{
  int pid1, pid2, child, status;
  printf("[%d] 부모 프로세스 시작 \n", getpid( ));
  pid1 = fork();
  if (pid1 == 0) {
    printf("[%d] 자식 프로세스[1] 시작 \n", getpid( ));
    sleep(1);
    printf("[%d] 자식 프로세스[1] 종료 \n", getpid( ));
    exit(1);
  }
  pid2 = fork();
  if (pid2 == 0) {
    printf("[%d] 자식 프로세스[2] 시작 \n", getpid( ));
    sleep(2);
    printf("[%d] 자식 프로세스[2] 종료 \n", getpid( ));
    exit(2);
  }
  child = waitpid(pid1, &status, 0); // 자식 프로세스 #1의 종료를 기다린다.
  printf("[%d] 자식 프로세스[1] %d 종료 \n", getpid(), child);
  printf("\t종료 코드 %d\n", status>>8);
  exit(0);
}

parent->pid1=fork  	      -> pid2=fork   ->  wait pid1
		ㄴ-> child1->exit ㄴ> child2->exit

```


```
pid, status = os.wait()
pid, status = os.waitpid(pid, options)
```

- `os.wait()`: 종료된 임의의 자식을 기다림
- `os.waitpid(pid, 0)`: 특정 자식이 종료될 때까지 기다림
- `os.WNOHANG`: 기다리지 않고 종료 여부만 확인
- 종료 상태는 `os.WIFEXITED()`, `os.WEXITSTATUS()` 등으로 해석


```py
import os
import time
import sys

def main():
    print(f"[{os.getpid()}] 부모 프로세스 시작")

    # 첫 번째 자식 프로세스 생성
    pid1 = os.fork()

    if pid1 == 0:
        # 자식 프로세스 [1]의 로직
        print(f"[{os.getpid()}] 자식 프로세스[1] 시작")
        time.sleep(1)
        print(f"[{os.getpid()}] 자식 프로세스[1] 종료")
        sys.exit(1)

    # 두 번째 자식 프로세스 생성
    pid2 = os.fork()

    if pid2 == 0:
        # 자식 프로세스 [2]의 로직
        print(f"[{os.getpid()}] 자식 프로세스[2] 시작")
        time.sleep(2)
        print(f"[{os.getpid()}] 자식 프로세스[2] 종료")
        sys.exit(2)

    # 자식 프로세스 #1의 종료를 기다린다.
    # os.waitpid(pid, options)는 (pid, status) 튜플을 반환합니다.
    child_pid, status = os.waitpid(pid1, 0)

    print(f"[{os.getpid()}] 자식 프로세스[1] {child_pid} 종료")
    
    # C언어의 status >> 8과 동일하게 종료 상태 값을 추출합니다.
    exit_code = os.waitstatus_to_exitcode(status)
    print(f"\t종료 코드 {exit_code}")

    sys.exit(0)

if __name__ == "__main__":
    main()
```
