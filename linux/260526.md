```py
import signal
import time
import sys

def alarm_handler(signum, frame):
    print("timeout")
    sys.exit(0)

signal.signal(signal.SIGALRM, alarm_handler)
signal.alarm(5)

sec = 0
while True:
    time.sleep(1)
    sec += 1
    print(f"{sec}초 경과")
```
